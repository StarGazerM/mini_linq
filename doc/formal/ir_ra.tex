\section{RA prime: named relational algebra with Rust expressions}
\label{sec:ra}

\paragraph{Grammar.} A standalone named DAG; declarations and output are part of it.
\begin{lstlisting}
RA ::= relational { Decl* (r = Op;)* output r as Head; }
Op ::= unit | rename R {source -> variable, ...} [constants {source = literal, ...}]
     | natural_join l with r | semijoin l with r
     | project r keep {columns} | select r if (Theta)
     | guard r if (Expr) [Environment]
     | bind r with [for] Pat [: Type] = (Expr) [Environment]
     | union r1, r2, ... [BlockOrigin] | anti [F] l, r
Environment ::= scope[names]
BlockOrigin ::= block free[F] interface[I]
                [fallback] [over prefix] [domain input]
\end{lstlisting}
A \code{rename} names a declared relation; its identity is its result name.
Its mapping and literal bindings partition the declared columns. Writing
$C$ for those bindings, the rename denotes
$\delta_m(\pi_{\dom m}\sigma_C(R))$; the dropped columns are not interface columns
and cannot prove disjointness of union arms.
Mathematically,
$o::=\mathbf1\mid\delta_m(\pi_{\dom m}\sigma_C(R))\mid r\bowtie s\mid r\ltimes s\mid\pi_Ur\mid\sigma_\theta r
\mid\operatorname{guard}_er\mid\operatorname{bind}_{N,e}r
\mid\operatorname{for}_{N,e}r\mid\bigcup_jr_j\mid r\triangleright_Fs$,
and $\Delta(r)=(V,T,C)$ records sort, symbolic Rust types and computed columns.
The optional \code{for} token distinguishes set-valued unnest from scalar
extension within the same Bind operator; its absence preserves scalar meaning.
The Environment syntax is unchanged. IX and Physical preserve this token.

\paragraph{Contract.}
\begin{mathpar}
\Rule{RA-Unit}{}{\Sigma;\Delta\vdash\mathbf1:\emptyset}

\Rule{RA-Rename}{R\in\dom\Sigma\quad
 \operatorname{cols}(\Sigma(R))=\dom m\uplus\dom C\quad m:\dom m\cong V\\
 C\text{ assigns admissible literals to declared columns}}
 {\Sigma;\Delta\vdash\delta_m(\pi_{\dom m}\sigma_C(R)):V}

\Rule{RA-Join}{\Delta(r).V=V\quad\Delta(s).V=W}
 {\Sigma;\Delta\vdash r\bowtie s:V\cup W}

\Rule{RA-Semi}{\Delta(r).V=V\quad s\in\dom\Delta}
 {\Sigma;\Delta\vdash r\ltimes s:V}

\Rule{RA-Project}{\Delta(r).V=V\quad U\subseteq V}
 {\Sigma;\Delta\vdash\pi_Ur:U}

\Rule{RA-Select}{\Delta(r).V=V\quad \theta\checkmark_{V,P}}
 {\Sigma;\Delta\vdash\sigma_\theta r:V}

\Rule{RA-Extend}{\Delta(r).V=V\quad e\checkmark_V\quad N\cap V=\emptyset}
 {\Sigma;\Delta\vdash\operatorname{bind}_{N,e}r:V\cup N}

\Rule{RA-For}{\Delta(r).V=V\quad e\checkmark_V\quad N=\set(\binds(p))\quad N\cap V=\emptyset}
 {\Sigma;\Delta\vdash\operatorname{for}_{N,e}r:V\cup N}

\Rule{RA-Union}{\forall j,\ H(r_j)=H}
 {H(\mathsf{union}(r_1,\ldots,r_n))=H\quad
 \sem{\mathsf{union}(r_1,\ldots,r_n)}=\bigcup_j\sem{r_j}}

\Rule{RA-Anti}{F\subseteq H(r)\cap H(s)}
 {H(\mathsf{anti}_F(r,s))=H(r)\quad
 \sem{\mathsf{anti}_F(r,s)}=\sem r\triangleright_F\sem s}

\Rule{RA-Def}{\Sigma;\Delta\vdash o:\delta\quad r\notin\dom\Delta}
 {\Sigma\vdash(\Delta,r=o)\Rightarrow\Delta[r\mapsto\delta]}

\Rule{WF-RA}{\mathsf{RAHeading}(a,A)}{\mathsf{WF}_{RA}(a)\Downarrow A}
\end{mathpar}
For the following clauses, write $a.defs=(r_i=o_i)_{i=1}^n$,
$H(r)=A(r).V$, and $\operatorname{reach}(r)$ for operand-graph reachability,
including $r$ itself. $\operatorname{names}$ extracts names from a sequence;
$\operatorname{FV}(e)$ is lexical Rust free-variable analysis (including
opaque macro dependencies as in the shared Rust analyzer).
$L$ is the set of all rename targets, bind names and aggregate names in $a$.
For a guard or bind $b$, $E_b$ is its optional Environment and $e_b$ its
expression. $\operatorname{scope}(b)$ is $H(b.input)$ without an Environment,
and $\set(E_b.variables)$ otherwise.
The optional-field notation $x=\bot$ means absent.
\begin{mathpar}
\Rule{RA-H-Declarations}{
 |\set(\operatorname{names}(a.decls))|=|a.decls|\quad
 \forall d\in a.decls,\ |d.columns|>0\ \land\
 |\set(\operatorname{names}(d.columns))|=|d.columns|\\
 \forall d\in a.decls,\ \forall h\in d.hosts,\
 \mathsf{hostReads}(d.name,h.key)=h.reads\ \land\
 \mathsf{hostSupports}(d.name,h.key)=h.supports\\
 \forall d\in a.decls,\ d.name=a.output.head.relation
 \Rightarrow(d.value=\bot\land d.hosts=[])\\
 |\{r_i\mid1\leq i\leq n\}|=n\quad
 \forall i,\ \operatorname{refs}(o_i)\subseteq\{r_j\mid j<i\}\\
 a.output.input\in\{r_i\mid1\leq i\leq n\}\quad
 \operatorname{reach}(a.output.input)=\{r_i\mid1\leq i\leq n\}
 }{\mathsf{RADeclarations}(a,A)}

\Rule{RA-H-Payloads}{
 \forall b\in\operatorname{guardBind}(a),\
 E_b=\bot\Rightarrow \operatorname{FV}(e_b)\cap L\subseteq H(b.input)\\
 \forall b\in\operatorname{guardBind}(a),\
 E_b\ne\bot\Rightarrow
 \bigl(|\set(E_b.variables)|=|E_b.variables|\ \land\
 \set(E_b.variables)\subseteq H(b.input)\bigr)
 }{\mathsf{RAPayloads}(a,A)}

\Rule{RA-H-Unions}{
 \forall u\in\operatorname{unions}(a),\ |u.operands|\geq2\ \land\
 \forall r,s\in u.operands,\ H(r)=H(s)\\
 \forall u\in\operatorname{unions}(a),\ \forall r,s\in u.operands,\ \forall x\in
 (\dom A(r).logicalTypeIds\setminus A(r).C)\cap
 (\dom A(s).logicalTypeIds\setminus A(s).C),\
 A(r).logicalTypeIds(x)=A(s).logicalTypeIds(x)\\
 \forall u\in\operatorname{unions}(a),\ u.origin\ne\bot\Rightarrow
 \bigl(|F_u|=|u.origin.free|\ \land\ |I_u|=|u.origin.interface|\ \land\
 F_u\cap I_u=\emptyset\\
 {}\land (u.origin.domain\ne\bot\Rightarrow
 (u.origin.over=\bot\land H(u.origin.domain)=F_u))\\
 {}\land (u.origin.over\ne\bot\Rightarrow
 (F_u\subseteq H(u.origin.over)\land I_u\cap H(u.origin.over)=\emptyset))\\
 {}\land H(u.operands_1)=F_u\cup I_u\cup P_u\bigr)
 }{\mathsf{RAUnions}(a,A)}

\Rule{RA-H-Output}{
 \forall e\in a.output.head.args,\ e\in\operatorname{Ident}\\
 \set(a.output.head.args)\setminus\operatorname{captureNames}(a)
 =H(a.output.input)
 }{\mathsf{RAOutput}(a,A)}

\Rule{RA-H-Rust}{
 \forall b\in\operatorname{guardBind}(a),\
 \operatorname{args}(b)=
 \{(x,\operatorname{column}(b.input,x))\mid
 x\in\operatorname{FV}(e_b)\cap\operatorname{scope}(b)\}
 }{\mathsf{RARust}(a,A)}

\Rule{RA-H-Heading}{
 \Delta_0=\emptyset\quad
 \forall i\in\{1,\ldots,n\},\
 \Sigma\vdash(\Delta_{i-1},r_i=o_i)\Rightarrow\Delta_i\quad A=\Delta_n\\
 \mathsf{RADeclarations}(a,A)\quad
 \mathsf{RAPayloads}(a,A)\quad\mathsf{RAUnions}(a,A)\quad
 \mathsf{RAOutput}(a,A)\quad\mathsf{RARust}(a,A)
 }{\mathsf{RAHeading}(a,A)}
\end{mathpar}
Here $\Sigma$ is the declaration map of $a$; $\operatorname{refs}$ includes
ordinary operands and block-origin domain/prefix references, while reachability
uses only ordinary operands, as the implementation does.
$F_u=\set(u.origin.free)$, $I_u=\set(u.origin.interface)$, and $P_u$ is
$H(u.origin.over)$ when present and $\emptyset$ otherwise.
$\operatorname{args}$ and $\operatorname{column}$ denote the structural
schedule's recorded Rust arguments and input column identities; RA-H-Rust
specifies their construction, not a second local rejection check.
The output rule reflects the standalone checker: computations belong in final
bind nodes, and output columns match the entire input heading after removing
host captures. Explicit Environments validate their listed scope;
they do not add a free-variable subset test on the expression.
Types are opaque host syntax; merging transports annotations and Rust
establishes concrete type compatibility.
Generator outputs are computed columns with item-derived types. A generator
does not inherit a scalar literal value or a scalar-bind uniqueness proof:
one input row may produce many distinct extensions (or none).

\paragraph{Meaning.} Section~\ref{sec:denotation}'s total operators over the
DAG; a name denotes an immutable set and promises neither materialization nor
a reusable iterator. Ordered fallback chooses its first nonempty arm.
Guards, scalar binds and generators are pure-total Kleisli steps.

\paragraph{Recorded decisions.} Join shape and association, where each relational
selection and semijoin sits, projections, and the block form (hoisted derived
relation or correlated arms over a prefix).

\paragraph{Rewrite check (\code{cq_lang_ra::check_rewrite}).} A structural contract
over renames identified by relation, mapping and constants (with their
ordinal among equal renames in definition order) and blocks identified by
their signature, never an equivalence decision:
(1)~both programs pass the contract; declarations, output heading and head are
unchanged;
(2)~the multiset of renames is equal and each keeps its column bindings;
(3)~relational constraints and projection correlations are equal after
union-find closure of equalities, and every source semijoin reduction survives
with its column matches;
(4)~the Rust payload multiset and each payload's free-variable input equality
classes are equal; binding origins are identified by payload rather than
written ordinal. There is no temporal relational-predecessor obligation;
(5)~union operand $j$ contains exactly branch $j$'s occurrences; a fallback
operand antijoins the earlier original operands by reference; a negated operand
contains exactly its branch, with no semijoin pushed into it.
The scalar/generator tag is part of the Rust node payload in every structural
certificate; changing only that tag is not a permitted rewrite.
This support certificate can establish equality independently of intermediate
evaluation domains. Pipeline refinements additionally preserve the recorded
VO join and predicate placement decisions; semantic equivalence alone does
not grant a later pass planning authority. No rename may be added or
removed (no query minimization).
